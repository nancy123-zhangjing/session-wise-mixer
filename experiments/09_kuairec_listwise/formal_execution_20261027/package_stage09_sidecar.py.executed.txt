def need(v,msg):
    if not v:raise RuntimeError(msg)
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def read(p):return json.loads(p.read_text(encoding='utf-8-sig'))
def save(p,v):
    with p.open('x',encoding='utf-8') as f:json.dump(v,f,ensure_ascii=False,indent=2);f.write('\n')
def call(args,timeout=240):
    r=subprocess.run([CLI,'--account','sii-shared',*args],capture_output=True,encoding='utf-8',errors='replace',timeout=timeout)
    need(r.returncode==0,'CLI operation failed; private output withheld');return r.stdout
def packet(code,marker='SIDECAR_PACKET '):
    raw=call(NB+['-c',code]);rows=[json.loads(x[len(marker):]) for x in raw.splitlines() if x.startswith(marker)]
    need(len(rows)==1,'No complete framed receipt');return rows[0]
def ref(local,remote):return dict(path=remote,sha256=sha(local))
def release_audit(name):return dict(job_name=name,all_allocated_pods_released=True,native_events_query_was_scoped_to_this_job=True,checked_at_utc=datetime.now(timezone.utc).isoformat(),auditor='root native scoped events and single-rank interval verification')

    identities={key:dict(sha256=sha(p),bytes=p.stat().st_size) for key,p in files.items()}
    check=f'''from pathlib import Path
import hashlib,json
